        function generateSensi(e) {
            e.preventDefault();
            const ram = parseInt(document.getElementById('ramSize').value);
            const phoneInputRaw = document.getElementById('phoneModel').value.trim();
            const phoneInput = phoneInputRaw.toLowerCase();
            
            let hash = 0;
            for (let i = 0; i < phoneInput.length; i++) {
                hash = phoneInput.charCodeAt(i) + ((hash << 5) - hash);
            }
            hash = Math.abs(hash);

            let baseMax;
            if (ram <= 3) baseMax = 200;
            else if (ram === 4) baseMax = 192;
            else if (ram === 6) baseMax = 180;
            else baseMax = 165;

            let gen = (baseMax - (hash % 15)); 
            let red = (baseMax - 5 - (hash % 18));
            let scope2 = (baseMax - 8 - (hash % 20));
            let scope4 = (baseMax - 15 - (hash % 25));
            let awm = (120 - (hash % 30));
            let btn = (40 + (hash % 22)) + '%';
            
            let dpi = (380 + (hash % 220)) + ' DPI';
            if (ram <= 3) dpi = (450 + (hash % 100)) + ' DPI';

            let shortName = phoneInputRaw.replace(/[^A-Z0-9a-z]/g, '').substring(0, 5).toUpperCase() || "DEV";
            let codeRandomness = Math.floor(10000 + (hash % 89999));
            const sensiCode = `FF-${shortName}-${ram}G-${codeRandomness}`;

            document.getElementById('valGen').innerText = gen;
            document.getElementById('valRed').innerText = red;
            document.getElementById('val2x').innerText = scope2;
            document.getElementById('val4x').innerText = scope4;
            document.getElementById('valAwm').innerText = awm;
            document.getElementById('valBtn').innerText = btn;
            document.getElementById('valDpi').innerText = dpi;

            const resultsDiv = document.getElementById('sensiResults');

            let codeBox = document.getElementById('generatedSensiCode');
            if(!codeBox) {
                codeBox = document.createElement('div');
                codeBox.id = 'generatedSensiCode';
                codeBox.className = 'col-span-2 md:col-span-4 mt-3 bg-slate-900 p-3 rounded-xl border border-pink-500/30 text-center shadow-lg';
                resultsDiv.appendChild(codeBox);
            }
            codeBox.innerHTML = `<span class="block text-[11px] text-pink-400 uppercase font-bold mb-1">Your Custom Optimized Device Code</span>
                                 <span class="text-xl font-black text-white font-mono select-all">${sensiCode}</span>
                                 <p class="text-[10px] text-slate-500 mt-1">Unique setup generated exclusively for ${phoneInputRaw}</p>`;

            resultsDiv.classList.remove('hidden');
            showToast("Settings uniquely optimized for your device! 🎯");
        }

        /* Multi-Step Checkout Modal */
        function openCheckout(productId) {
            currentSelectedItem = products.find(p => p.id === productId);
            if (!currentSelectedItem) return;

            document.getElementById('checkoutItemTitle').innerHTML = `Item: <strong class="text-white">${currentSelectedItem.title}</strong>`;
            document.getElementById('checkoutItemPrice').innerText = `₦${Number(currentSelectedItem.price).toLocaleString()}`;
            
            // Reset to step 1
            nextCheckoutStep(1);
            openModal('checkoutModal');
        }

        function nextCheckoutStep(step) {
            document.getElementById('checkoutStep1').classList.add('hidden');
            document.getElementById('checkoutStep2').classList.add('hidden');
            document.getElementById('checkoutStep3').classList.add('hidden');

            document.getElementById(`checkoutStep${step}`).classList.remove('hidden');

            const dots = ['dot1', 'dot2', 'dot3'];
            dots.forEach((d, i) => {
                const el = document.getElementById(d);
                if(i + 1 === step) {
                    el.className = "w-3 h-3 rounded-full bg-pink-500";
                } else if(i + 1 < step) {
                    el.className = "w-3 h-3 rounded-full bg-green-500";
                } else {
                    el.className = "w-3 h-3 rounded-full bg-slate-700";
                }
            });

            const texts = ["Step 1 of 3: Gamer Name", "Step 2 of 3: Player ID & Phone", "Step 3 of 3: Bank Payment & Receipt"];
            document.getElementById('stepIndicatorText').innerText = texts[step - 1];
        }

        function handleFileSelect(e) {
            const file = e.target.files[0];
            if (file) {
                document.getElementById('fileLabel').innerHTML = `<i class="fa-solid fa-circle-check text-green-400 text-lg"></i> <span class="text-green-400 font-bold">${file.name} Attached</span>`;
                const reader = new FileReader();

            e.preventDefault();
            if (!uploadedReceiptBase64) {
