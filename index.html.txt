    <script>
        function playSound(type = 'click') {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);

                if (type === 'click') {
                    osc.frequency.setValueAtTime(450, ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(800, ctx.currentTime + 0.05);
                    gain.gain.setValueAtTime(0.08, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.05);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.05);
                } else if (type === 'success') {
                    osc.frequency.setValueAtTime(587.33, ctx.currentTime);
                    osc.frequency.setValueAtTime(880, ctx.currentTime + 0.08);
                    gain.gain.setValueAtTime(0.1, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.25);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.25);
                }
            } catch (e) {
                // AudioContext not allowed before user gesture or unsupported
            }
        }
    </script>

        // Secret Admin Trigger: Click '.NG' or ' LIFE' 5 times quickly
        function initSecretAdminTrigger() {
            let clicks = 0;
            const trigger = document.getElementById('secretAdminTrigger');
            trigger.addEventListener('click', () => {
                clicks++;
                playSound('click');
                if (clicks >= 5) {
                    openModal('adminAuthModal');
                    clicks = 0;
                }
                setTimeout(() => { clicks = 0; }, 3000);
            });
        }

        function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
        function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

        function startLiveCounter() {
            setInterval(() => {
                const el = document.getElementById('heroOnlineCount');
                let curr = parseInt(el.innerText.replace(/,/g, ''));
                let diff = Math.floor(Math.random() * 25) - 10;
                curr += diff;
                if (curr > 120000) curr = 108500;
                if (curr < 100000) curr = 101200;
                el.innerText = curr.toLocaleString();
            }, 4000);
        }

        // Admin Auth & Dashboard

            e.preventDefault();
            const u = document.getElementById('adminUser').value.trim();
            const p = document.getElementById('adminPass').value.trim();

                playSound('success');
                closeModal('adminAuthModal');
                openModal('adminModal');
                
                // Load current values into inputs
                document.getElementById('editHeroHeading').value = document.getElementById('heroHeading').innerText;
